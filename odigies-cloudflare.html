🔗 Σύντομοι σύνδεσμοι καταλόγου

Γίνεται μία φορά από τον διαχειριστή της GlenApps. Εξυπηρετεί όλους τους πωλητές της εφαρμογής «Παραγγελίες». Δωρεάν (έως 100.000 ανοίγματα την ημέρα).

  1. Μπες στο dash.cloudflare.com και κάνε δωρεάν εγγραφή με το email σου. Αν σου ζητήσει κάρτα, σταμάτα και πες το.
  2. Αριστερά: Storage & Databases → D1 (ή «Αποθήκευση και βάσεις δεδομένων») → Create database. Όνομα: glenapps → Create.
  3. Αριστερά: Workers & Pages → Create → Create Worker (ή «Start with Hello World»). Όνομα: glenapps-k → Deploy.
  4. Πάτα Edit code. Σβήσε όλον τον κώδικα που υπάρχει, πάτα το κουμπί παρακάτω «📋 Αντιγραφή κώδικα» και επικόλλησε. Μετά Deploy.
  5. Γύρνα στο glenapps-k → Settings → Bindings → Add → D1 database. Variable name: DB · Database: glenapps → Save (ή Deploy).
  6. Άνοιξε τη διεύθυνση του glenapps-k (κάτι σαν https://glenapps-k.ΟΝΟΜΑ.workers.dev). Πρέπει να γράφει «λειτουργεί ✓» χωρίς προειδοποίηση.
  7. Στείλε αυτή τη διεύθυνση στο Claude για να μπει στην εφαρμογή.

/* GlenApps · Παραγγελίες — σύντομοι σύνδεσμοι καταλόγου
   Φυλάει τον (υπογεγραμμένο) κατάλογο κάθε πωλητή και δίνει έναν κωδικό 7 γραμμάτων.
   Ο πελάτης ανοίγει «…/p.html#s=κωδικός» και η σελίδα φέρνει τον κατάλογο από εδώ.
   Δεν περνούν ποτέ από εδώ στοιχεία πελατών ή παραγγελίες.
   Χρειάζεται μία βάση D1 συνδεδεμένη με όνομα μεταβλητής DB. */
const ALLOW = ["https://aglentzakis-dot.github.io", "http://localhost:8765"];
const AL = "abcdefghijkmnpqrstuvwxyz23456789";
const MAX = 400000;
const H = { "access-control-allow-origin": "*", "access-control-allow-methods": "GET,POST,OPTIONS", "access-control-allow-headers": "content-type", "access-control-max-age": "86400" };
const J = (o, st = 200, x = {}) => new Response(JSON.stringify(o), { status: st, headers: { ...H, "content-type": "application/json; charset=utf-8", ...x } });
const dec = t => { t = String(t).replace(/-/g, "+").replace(/[_.]/g, "/"); while (t.length % 4) t += "="; const s = atob(t), u = new Uint8Array(s.length); for (let i = 0; i < s.length; i++) u[i] = s.charCodeAt(i); return u; };
async function signed(sp, pk, g) {
  try {
    const k = await crypto.subtle.importKey("raw", dec(sp), { name: "ECDSA", namedCurve: "P-256" }, false, ["verify"]);
    return await crypto.subtle.verify({ name: "ECDSA", hash: "SHA-256" }, k, dec(g), new TextEncoder().encode(pk));
  } catch { return false; }
}
let ready = false;
async function init(db) {
  if (ready) return;
  await db.prepare("CREATE TABLE IF NOT EXISTS cat (id TEXT PRIMARY KEY, sp TEXT NOT NULL, pk TEXT NOT NULL, g TEXT NOT NULL, ts INTEGER NOT NULL)").run();
  ready = true;
}
const newId = () => [...crypto.getRandomValues(new Uint8Array(7))].map(x => AL[x % AL.length]).join("");
export default {
  async fetch(req, env) {
    const u = new URL(req.url);
    if (req.method === "OPTIONS") return new Response(null, { headers: H });
    if (u.pathname === "/") return new Response("GlenApps · σύντομοι σύνδεσμοι καταλόγου: λειτουργεί ✓" + (env.DB ? "" : "\n⚠️ Λείπει η βάση D1 με όνομα DB"), { headers: { ...H, "content-type": "text/plain; charset=utf-8" } });
    if (!env.DB) return J({ err: "Λείπει η βάση D1 (DB)" }, 500);
    await init(env.DB);
    const m = u.pathname.match(/^\/c\/([a-z0-9]{7})$/);
    if (req.method === "GET" && m) {
      const r = await env.DB.prepare("SELECT pk, g, sp, ts FROM cat WHERE id = ?").bind(m[1]).first();
      return r ? J(r, 200, { "cache-control": "public, max-age=30" }) : J({ err: "Δεν βρέθηκε" }, 404);
    }
    if (req.method === "POST" && u.pathname === "/c") {
      const o = req.headers.get("origin") || ""; if (o && !ALLOW.includes(o)) return J({ err: "Μη επιτρεπτή προέλευση" }, 403);
      if ((+req.headers.get("content-length") || 0) > MAX + 2000) return J({ err: "Πολύ μεγάλος" }, 413);
      let b; try { b = await req.json(); } catch { return J({ err: "Λάθος δεδομένα" }, 400); }
      const { pk, g, sp } = b || {}; let id = b && b.id;
      if (typeof pk !== "string" || typeof g !== "string" || typeof sp !== "string" || pk.length > MAX || sp.length > 120 || g.length > 200) return J({ err: "Λάθος δεδομένα" }, 400);
      if (!(await signed(sp, pk, g))) return J({ err: "Λάθος υπογραφή" }, 403);
      const now = Date.now();
      if (typeof id === "string" && /^[a-z0-9]{7}$/.test(id)) {
        const r = await env.DB.prepare("SELECT sp FROM cat WHERE id = ?").bind(id).first();
        if (r && r.sp !== sp) return J({ err: "Ο κωδικός ανήκει σε άλλον πωλητή" }, 403);
        if (r) { await env.DB.prepare("UPDATE cat SET pk = ?, g = ?, ts = ? WHERE id = ?").bind(pk, g, now, id).run(); return J({ id }); }
      }
      for (let i = 0; i < 6; i++) {
        id = newId();
        try { await env.DB.prepare("INSERT INTO cat (id, sp, pk, g, ts) VALUES (?, ?, ?, ?, ?)").bind(id, sp, pk, g, now).run(); return J({ id }); } catch {}
      }
      return J({ err: "Δεν δόθηκε κωδικός" }, 500);
    }
    return J({ err: "Δεν υπάρχει" }, 404);
  }
};

Στην υπηρεσία φυλάγεται μόνο ο κατάλογος κάθε πωλητή (στοιχεία καταστήματος, προϊόντα, τιμές), υπογεγραμμένος με το κλειδί του. Στοιχεία πελατών και παραγγελίες δεν περνούν ποτέ από εκεί.
